cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(comdrivers_display LANGUAGES C)
endif()

# 此demo自包含，不调用兄弟目录或仓库公共目录。
file(GLOB_RECURSE DEMO_SOURCES CONFIGURE_DEPENDS Middleware/*.c Service/*.c)
list(FILTER DEMO_SOURCES EXCLUDE REGEX "/Lvgl8/")
add_library(comdrivers_display STATIC ${DEMO_SOURCES})
target_include_directories(comdrivers_display PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_features(comdrivers_display PUBLIC c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(comdrivers_display PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(comdrivers_display PUBLIC m)
endif()

option(COMDRIVERS_ENABLE_STM32F4 "编译所选demo的STM32F4映射" OFF)
set(COMDRIVERS_STM32_INCLUDE_DIRS "" CACHE STRING "HAL、CMSIS、HAL配置头文件目录")
set(COMDRIVERS_STM32_DEVICE "STM32F407xx" CACHE STRING "STM32F4芯片型号宏")
if(COMDRIVERS_ENABLE_STM32F4)
    target_sources(comdrivers_display PRIVATE Mapping/STM32F4/stm32f4_mapping.c)
    target_include_directories(comdrivers_display PRIVATE ${COMDRIVERS_STM32_INCLUDE_DIRS})
    target_compile_definitions(comdrivers_display PRIVATE USE_HAL_DRIVER ${COMDRIVERS_STM32_DEVICE})
endif()

option(COMDRIVERS_ENABLE_LVGL8 "编译DISPLAY_demo中的LVGL8适配" OFF)
set(COMDRIVERS_LVGL_INCLUDE_DIRS "" CACHE STRING "LVGL与lv_conf.h目录")
if(COMDRIVERS_ENABLE_LVGL8)
    target_sources(comdrivers_display PRIVATE Service/Lvgl8/lvgl8_adapter.c)
    target_include_directories(comdrivers_display PRIVATE ${COMDRIVERS_LVGL_INCLUDE_DIRS})
endif()

# 测试留在本demo内部，不成为固件源码依赖。
option(COMDRIVERS_BUILD_TESTS "构建当前demo的回归测试" OFF)
if(COMDRIVERS_BUILD_TESTS)
    enable_testing()
    add_executable(comdrivers_display_regression Tests/test_regression.c)
    target_link_libraries(comdrivers_display_regression PRIVATE comdrivers_display)
    if(NOT CMAKE_CROSSCOMPILING)
        add_test(NAME comdrivers_display_regression COMMAND comdrivers_display_regression)
    endif()
endif()
